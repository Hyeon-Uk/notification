CMAKE_MINIMUM_REQUIRED(VERSION 3.12)

PROJECT(notification)

SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall")
IF(NOT ASAN_ENABLED)
  SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Werror")
ENDIF(NOT ASAN_ENABLED)
SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wno-error=deprecated-declarations")
SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wl,-zdefs")
SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fvisibility=hidden")
SET(CMAKE_C_FLAGS_DEBUG "-O0 -g")
SET(CMAKE_C_FLAGS_RELEASE "-O2")

SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${EXTRA_C_FLAGS} -std=c++23")
SET(CMAKE_CXX_FLAGS_DEBUG "-O0 -g")
SET(CMAKE_CXX_FLAGS_RELEASE "-O2")

# Phase 4 / 4.6: optional LTO (link-time optimization).  Default
# OFF so Tizen distros are unaffected; size-sensitive builds can
# enable with -DENABLE_LTO=ON at CMake configure time.  Applies
# to all C and C++ TUs and to both shared and static linker
# invocations.  Pairs naturally with the gc-sections work in
# 4.1 (per-target compile/link options).
OPTION(ENABLE_LTO "Enable link-time optimization (-flto)" OFF)
IF(ENABLE_LTO)
  SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -flto")
  SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -flto")
  SET(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -flto")
  SET(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -flto")
  MESSAGE(STATUS "Phase 4 / 4.6: LTO enabled")
ENDIF(ENABLE_LTO)

SET(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH}
  "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Modules/")

SET(TARGET_NOTIFICATION "notification")
SET(TARGET_NOTIFICATION_EX "notification-ex")
SET(TARGET_NOTIFICATION_INIT "notification_init")
SET(TARGET_NOTIFICATION_TEST_APP "notification-test-app")
SET(TARGET_NOTIFICATION_UNITTESTS "notification-unittests")
SET(TARGET_NOTIFICATION_SMOKE "notification-smoke")
SET(TARGET_NOTIFICATION_INTERNAL_STATIC "notification-internal-static")
SET(TARGET_NOTIFICATION_INTERNAL_UNITTESTS "notification-internal-unittests")
SET(TARGET_NOTIFICATION_INTEGRATION_TESTS "notification-integration-tests")

  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke)

ADD_TEST(NAME ${TARGET_NOTIFICATION_INTEGRATION_TESTS}
  COMMAND ${TARGET_NOTIFICATION_INTEGRATION_TESTS}
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration)
