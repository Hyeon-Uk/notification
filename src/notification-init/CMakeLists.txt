AUX_SOURCE_DIRECTORY(${CMAKE_CURRENT_SOURCE_DIR} INIT_SRCS)

ADD_EXECUTABLE(${TARGET_NOTIFICATION_INIT} ${INIT_SRCS})

TARGET_INCLUDE_DIRECTORIES(${TARGET_NOTIFICATION_INIT} PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/../
  ${CMAKE_CURRENT_SOURCE_DIR}/../notification
  ${CMAKE_CURRENT_SOURCE_DIR}/../notification/include
)

APPLY_PKG_CONFIG(${TARGET_NOTIFICATION_INIT} PUBLIC
  BUNDLE_DEPS
  CAPI_APPFW_APP_CONTROL_DEPS
  CAPI_APPFW_PACKAGE_MANAGER_DEPS
  INIPARSER_DEPS
  PKGMGR_INFO_DEPS
  VCONF_DEPS
)

SET_TARGET_PROPERTIES(${TARGET_NOTIFICATION_INIT} PROPERTIES
  COMPILE_FLAGS "${CMAKE_CXX_FLAGS} -fPIE")
SET_TARGET_PROPERTIES(${TARGET_NOTIFICATION_INIT} PROPERTIES
  LINK_FLAGS "-pie")

TARGET_LINK_LIBRARIES(${TARGET_NOTIFICATION_INIT} PRIVATE
  ${TARGET_NOTIFICATION})

INSTALL(TARGETS ${TARGET_NOTIFICATION_INIT} DESTINATION bin)
