SET(TARGET_NOTIFICATION_SMOKE "notification-smoke")

AUX_SOURCE_DIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/src SMOKE_SRCS)
AUX_SOURCE_DIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/../mock MOCK_SRCS)

ADD_EXECUTABLE(${TARGET_NOTIFICATION_SMOKE}
  ${SMOKE_SRCS}
  ${MOCK_SRCS}
)

TARGET_INCLUDE_DIRECTORIES(${TARGET_NOTIFICATION_SMOKE} PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/../
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification/include
  ${CMAKE_CURRENT_SOURCE_DIR}/../mock
)

APPLY_PKG_CONFIG(${TARGET_NOTIFICATION_SMOKE} PUBLIC
  AUL_DEPS
  CAPI_APPFW_APP_CONTROL_DEPS
  DLOG_DEPS
  GIO_DEPS
  GIO_UNIX_DEPS
  GLIB_DEPS
  GMOCK_DEPS
  LIBTZPLATFORM_CONFIG_DEPS
  PKGMGR_INFO_DEPS
  RPC_PORT_DEPS
  SECURITY_MANAGER_DEPS
  TIZEN_CORE_DEPS
  VCONF_DEPS
)

# Link only the in-scope library (NOT notification-ex). This proves the
# Phase-2 deprecation of notification-ex doesn't pull in transitively.
TARGET_LINK_LIBRARIES(${TARGET_NOTIFICATION_SMOKE} PRIVATE
  ${TARGET_NOTIFICATION} "-ldl")

INSTALL(TARGETS ${TARGET_NOTIFICATION_SMOKE} DESTINATION bin)
