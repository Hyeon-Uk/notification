AUX_SOURCE_DIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/src SMOKE_SOURCES)
SET(SMOKE_COMMON ${CMAKE_CURRENT_SOURCE_DIR}/common/smoke_utils.cc)

ADD_EXECUTABLE(${TARGET_NOTIFICATION_SMOKETESTS}
  ${SMOKE_SOURCES}
  ${SMOKE_COMMON}
)

TARGET_INCLUDE_DIRECTORIES(${TARGET_NOTIFICATION_SMOKETESTS} PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification/include
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification-ex
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification-ex/api
)

TARGET_LINK_LIBRARIES(${TARGET_NOTIFICATION_SMOKETESTS} PUBLIC
  ${TARGET_NOTIFICATION}
  "-ldl"
  "-lpthread"
)

SET_TARGET_PROPERTIES(${TARGET_NOTIFICATION_SMOKETESTS} PROPERTIES
  COMPILE_FLAGS "-fPIE"
)
SET_TARGET_PROPERTIES(${TARGET_NOTIFICATION_SMOKETESTS} PROPERTIES
  LINK_FLAGS "-pie"
)

APPLY_PKG_CONFIG(${TARGET_NOTIFICATION_SMOKETESTS} PUBLIC
  AUL_DEPS
  BUNDLE_DEPS
  DLOG_DEPS
  GIO_DEPS
  GIO_UNIX_DEPS
  GLIB_DEPS
  GMOCK_DEPS
  LIBTZPLATFORM_CONFIG_DEPS
  RPC_PORT_DEPS
  SQLITE3_DEPS
  TIZEN_CORE_DEPS
)

SET(SMOKE_PKGID org.tizen.notification.smoketest)
SET(SMOKE_APPDIR /usr/apps/${SMOKE_PKGID})
SET(SMOKE_MANIFESTDIR /usr/share/packages)

INSTALL(TARGETS ${TARGET_NOTIFICATION_SMOKETESTS}
        DESTINATION ${SMOKE_APPDIR}/bin/)
INSTALL(FILES ${CMAKE_CURRENT_SOURCE_DIR}/tizen-manifest.xml
        DESTINATION ${SMOKE_APPDIR}/)
INSTALL(FILES ${CMAKE_CURRENT_SOURCE_DIR}/tizen-manifest.xml
        DESTINATION ${SMOKE_MANIFESTDIR}/
        RENAME ${SMOKE_PKGID}.xml)
