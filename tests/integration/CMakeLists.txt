SET(TARGET_NOTIFICATION_INTEGRATION_TESTS "notification-integration-tests")

AUX_SOURCE_DIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/src INTEGRATION_SRCS)

ADD_EXECUTABLE(${TARGET_NOTIFICATION_INTEGRATION_TESTS}
  ${INTEGRATION_SRCS}
)

TARGET_INCLUDE_DIRECTORIES(${TARGET_NOTIFICATION_INTEGRATION_TESTS} PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification/include
)

APPLY_PKG_CONFIG(${TARGET_NOTIFICATION_INTEGRATION_TESTS} PUBLIC
  AUL_DEPS
  CAPI_APPFW_APP_CONTROL_DEPS
  DLOG_DEPS
  GIO_DEPS
  GIO_UNIX_DEPS
  GLIB_DEPS
  GMOCK_DEPS
  LIBTZPLATFORM_CONFIG_DEPS
  PKGMGR_INFO_DEPS
  RPC_PORT_DEPS
  SECURITY_MANAGER_DEPS
  SQLITE3_DEPS
  TIZEN_CORE_DEPS
  VCONF_DEPS
)

TARGET_LINK_LIBRARIES(${TARGET_NOTIFICATION_INTEGRATION_TESTS} PRIVATE
  ${TARGET_NOTIFICATION} "-ldl -lsqlite3")

INSTALL(TARGETS ${TARGET_NOTIFICATION_INTEGRATION_TESTS} DESTINATION bin)

ADD_TEST(NAME ${TARGET_NOTIFICATION_INTEGRATION_TESTS}
  COMMAND ${TARGET_NOTIFICATION_INTEGRATION_TESTS}
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
