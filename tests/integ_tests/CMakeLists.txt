AUX_SOURCE_DIRECTORY(${CMAKE_CURRENT_SOURCE_DIR}/src SOURCES)

ADD_EXECUTABLE(${TARGET_NOTIFICATION_INTEGTESTS}
  ${SOURCES}
)

TARGET_INCLUDE_DIRECTORIES(${TARGET_NOTIFICATION_INTEGTESTS} PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification/include
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification-ex
  ${CMAKE_CURRENT_SOURCE_DIR}/../../src/notification-ex/api
)

TARGET_LINK_LIBRARIES(${TARGET_NOTIFICATION_INTEGTESTS} PUBLIC
  ${TARGET_NOTIFICATION}
  "-ldl"
)

SET_TARGET_PROPERTIES(${TARGET_NOTIFICATION_INTEGTESTS} PROPERTIES
  COMPILE_FLAGS "-fPIE"
)
SET_TARGET_PROPERTIES(${TARGET_NOTIFICATION_INTEGTESTS} PROPERTIES
  LINK_FLAGS "-pie"
)

APPLY_PKG_CONFIG(${TARGET_NOTIFICATION_INTEGTESTS} PUBLIC
  AUL_DEPS
  BUNDLE_DEPS
  DLOG_DEPS
  GIO_DEPS
  GIO_UNIX_DEPS
  GLIB_DEPS
  GMOCK_DEPS
  LIBTZPLATFORM_CONFIG_DEPS
  RPC_PORT_DEPS
  SQLITE3_DEPS
  TIZEN_CORE_DEPS
)


SET(INTEGTEST_PKGID org.tizen.notification.integtest)
SET(INTEGTEST_APPDIR /usr/apps/${INTEGTEST_PKGID})
SET(INTEGTEST_MANIFESTDIR /usr/share/packages)

INSTALL(TARGETS ${TARGET_NOTIFICATION_INTEGTESTS}
        DESTINATION ${INTEGTEST_APPDIR}/bin/)
INSTALL(FILES ${CMAKE_CURRENT_SOURCE_DIR}/tizen-manifest.xml
        DESTINATION ${INTEGTEST_APPDIR}/)
INSTALL(FILES ${CMAKE_CURRENT_SOURCE_DIR}/tizen-manifest.xml
        DESTINATION ${INTEGTEST_MANIFESTDIR}/
        RENAME ${INTEGTEST_PKGID}.xml)
